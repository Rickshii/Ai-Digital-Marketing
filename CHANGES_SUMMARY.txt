BROKEN CODE (What was breaking login):
────────────────────────────────────────
let API_URL;
if (import.meta.env.VITE_API_URL) {
  API_URL = import.meta.env.VITE_API_URL;
} else if (import.meta.env.PROD) {
  const hostname = window.location.hostname;
